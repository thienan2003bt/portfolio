import type { Metadata } from "next";
import { notFound } from "next/navigation";
import { projects } from "@/content/projects";

type Props = { params: Promise<{ slug: string }> };

// A complete App Router page backed by the portfolio's typed content.
// Place this file at app/projects/[slug]/page.tsx when adopting the example.
export const dynamicParams = false;

export function generateStaticParams() {
  return projects.map(({ slug }) => ({ slug }));
}

function findProject(slug: string) {
  const project = projects.find((entry) => entry.slug === slug);
  if (!project) notFound();
  return project;
}

export async function generateMetadata({ params }: Props): Promise<Metadata> {
  const { slug } = await params;
  const project = findProject(slug);
  return { title: project.title, description: project.summary };
}

export default async function ProjectPage({ params }: Props) {
  const { slug } = await params;
  const project = findProject(slug);

  return (
    <main>
      <article>
        <header>
          <p>{project.role} · {project.year}</p>
          <h1>{project.title}</h1>
          <p>{project.summary}</p>
        </header>
        <h2>Outcome</h2>
        <p>{project.outcome}</p>
        <h2>Technologies</h2>
        <ul>{project.tags.map((tag) => <li key={tag}>{tag}</li>)}</ul>
        <nav aria-label="Project resources">
          {project.links.live && <a href={project.links.live}>Live demo</a>}
          {project.links.github && <a href={project.links.github}>Source code</a>}
        </nav>
      </article>
    </main>
  );
}
